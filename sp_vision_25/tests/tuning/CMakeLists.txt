cmake_minimum_required(VERSION 3.16)
project(sp_tuning_regression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(fmt REQUIRED)
find_package(spdlog REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
find_package(Threads REQUIRED)
find_library(SP_CORE sp_sim_core PATHS "${WORKSPACE}/sp_vision_ros/build-ros-ascii" NO_DEFAULT_PATH REQUIRED)
find_library(SP_TINY tinympcstatic PATHS "${WORKSPACE}/sp_vision_ros/build-ros-ascii/tinympc" NO_DEFAULT_PATH REQUIRED)
add_executable(tracker_interval_test tracker_interval_test.cpp)
target_include_directories(tracker_interval_test PRIVATE "${WORKSPACE}/sp_vision_25" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(tracker_interval_test PRIVATE ${SP_CORE} ${SP_TINY} Eigen3::Eigen ${OpenCV_LIBS}
  yaml-cpp fmt::fmt spdlog::spdlog openvino::runtime Threads::Threads)
add_executable(tracker_geometry_probe tracker_geometry_probe.cpp)
target_link_libraries(tracker_geometry_probe PRIVATE ${SP_CORE} ${SP_TINY} Eigen3::Eigen ${OpenCV_LIBS}
  yaml-cpp fmt::fmt spdlog::spdlog openvino::runtime Threads::Threads)
target_include_directories(tracker_geometry_probe PRIVATE "${WORKSPACE}/sp_vision_25" ${OpenCV_INCLUDE_DIRS})
enable_testing()
add_test(NAME tracker_intervals COMMAND tracker_interval_test "${WORKSPACE}/sp_vision_ros/config/simulator.yaml")
add_executable(fire_window_test fire_window_test.cpp)
target_include_directories(fire_window_test PRIVATE "${WORKSPACE}/sp_vision_25" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(fire_window_test PRIVATE ${SP_CORE} ${SP_TINY} Eigen3::Eigen ${OpenCV_LIBS}
  yaml-cpp fmt::fmt spdlog::spdlog openvino::runtime Threads::Threads)
add_test(NAME fire_window COMMAND fire_window_test "${WORKSPACE}/sp_vision_ros/config/simulator.yaml")
add_executable(inference_reuse_test inference_reuse_test.cpp)
target_compile_definitions(inference_reuse_test PRIVATE SP_SOURCE_ROOT="${WORKSPACE}/sp_vision_25")
target_include_directories(inference_reuse_test PRIVATE "${WORKSPACE}/sp_vision_25" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(inference_reuse_test PRIVATE ${SP_CORE} ${SP_TINY} Eigen3::Eigen ${OpenCV_LIBS}
  yaml-cpp fmt::fmt spdlog::spdlog openvino::runtime Threads::Threads)
