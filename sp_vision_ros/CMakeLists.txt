cmake_minimum_required(VERSION 3.16)
project(sp_vision_ros LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SP_VISION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../sp_vision_25" CACHE PATH "Existing SP source checkout (read only)")
option(SP_BUILD_ROS "Build ROS2 adapter" ON)
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(fmt REQUIRED)
find_package(spdlog REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenVINO REQUIRED COMPONENTS Runtime)
find_package(Threads REQUIRED)
find_package(nlohmann_json REQUIRED)
add_subdirectory(${SP_VISION_ROOT}/tasks/auto_aim/planner/tinympc tinympc)
target_link_libraries(tinympcstatic PUBLIC Eigen3::Eigen)
set(SP_SOURCES
  tasks/auto_aim/armor.cpp tasks/auto_aim/classifier.cpp tasks/auto_aim/detector.cpp
  tasks/auto_aim/solver.cpp tasks/auto_aim/target.cpp tasks/auto_aim/tracker.cpp
  tasks/auto_aim/yolo.cpp tasks/auto_aim/yolos/yolov5.cpp
  tasks/auto_aim/yolos/yolov8.cpp tasks/auto_aim/yolos/yolo11.cpp
  tasks/auto_aim/planner/planner.cpp tools/extended_kalman_filter.cpp
  tools/img_tools.cpp tools/math_tools.cpp tools/trajectory.cpp tools/logger.cpp)
list(TRANSFORM SP_SOURCES PREPEND "${SP_VISION_ROOT}/")
add_library(sp_sim_core STATIC ${SP_SOURCES})
target_include_directories(sp_sim_core PUBLIC ${SP_VISION_ROOT} ${CMAKE_CURRENT_SOURCE_DIR}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(sp_sim_core PUBLIC Eigen3::Eigen ${OpenCV_LIBS} yaml-cpp fmt::fmt spdlog::spdlog openvino::runtime tinympcstatic Threads::Threads)
add_executable(sp_sim_geometry_test tests/geometry_test.cpp)
target_link_libraries(sp_sim_geometry_test sp_sim_core)
add_executable(sp_sim_image_probe tests/image_probe.cpp)
target_compile_definitions(sp_sim_image_probe PRIVATE SP_SOURCE_ROOT="${SP_VISION_ROOT}")
target_link_libraries(sp_sim_image_probe sp_sim_core nlohmann_json::nlohmann_json)
add_executable(sp_sim_ipc ../sp_vision_sim/src/main.cpp)
target_include_directories(sp_sim_ipc PRIVATE ../sp_vision_sim/include)
target_compile_definitions(sp_sim_ipc PRIVATE SP_SOURCE_ROOT="${SP_VISION_ROOT}")
target_link_libraries(sp_sim_ipc sp_sim_core nlohmann_json::nlohmann_json)
enable_testing()
add_test(NAME geometry_and_planner COMMAND sp_sim_geometry_test ${CMAKE_CURRENT_SOURCE_DIR}/configs/standard3.yaml)
if(SP_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(tf2_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(rm_interfaces REQUIRED)
  add_executable(sp_sim_auto_aim src/ros_node.cpp)
  target_compile_definitions(sp_sim_auto_aim PRIVATE SP_SOURCE_ROOT="${SP_VISION_ROOT}")
  target_link_libraries(sp_sim_auto_aim sp_sim_core nlohmann_json::nlohmann_json)
  ament_target_dependencies(sp_sim_auto_aim rclcpp sensor_msgs tf2_msgs std_msgs rm_interfaces)
  install(TARGETS sp_sim_auto_aim DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY configs launch DESTINATION share/${PROJECT_NAME})
  ament_package()
endif()
