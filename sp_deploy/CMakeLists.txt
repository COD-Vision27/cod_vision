cmake_minimum_required(VERSION 3.16)
project(sp_real_deploy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SP_VISION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../sp_vision_25" CACHE PATH "Read-only original SP project")
set(OpenVINO_DIR "/opt/intel/openvino_2024.6.0/runtime/cmake" CACHE PATH "OpenVINO development package")
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(fmt REQUIRED)
find_package(spdlog REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
set(SP_IO_OVERLAY_ROOT "${CMAKE_CURRENT_BINARY_DIR}/sp-overlay" CACHE PATH "Build-only fresh-pose IO overlay")
if(NOT EXISTS "${SP_IO_OVERLAY_ROOT}/io/gimbal/gimbal.hpp")
  message(FATAL_ERROR "Run deploy.py build to generate the read-only source overlay")
endif()
include_directories(BEFORE ${SP_IO_OVERLAY_ROOT} ${SP_VISION_ROOT} ${EIGEN3_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
# Reuse the already verified original YOLO/Solver/Tracker/Planner source selection.
set(SP_BUILD_ROS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../sp_vision_ros" sp-core)
add_subdirectory("${SP_IO_OVERLAY_ROOT}/io" sp-io)
add_executable(sp_capture src/capture.cpp)
target_link_libraries(sp_capture io sp_sim_core nlohmann_json::nlohmann_json)
add_executable(sp_real_aim src/real_aim.cpp)
target_link_libraries(sp_real_aim io sp_sim_core nlohmann_json::nlohmann_json)
add_executable(sp_deploy_protocol_test "${SP_VISION_ROOT}/tests/gimbal_protocol_test.cpp")
target_link_libraries(sp_deploy_protocol_test io sp_sim_core util)
enable_testing()
add_test(NAME original_gimbal_protocol COMMAND sp_deploy_protocol_test)
set_tests_properties(original_gimbal_protocol PROPERTIES TIMEOUT 10)
add_test(NAME real_control_policy COMMAND sp_real_aim --self-test)

add_executable(sp_live_pose_test tests/live_pose_test.cpp)
target_link_libraries(sp_live_pose_test io sp_sim_core util)
add_test(NAME live_received_pose COMMAND sp_live_pose_test)
set_tests_properties(live_received_pose PROPERTIES TIMEOUT 10)
# Every IO consumer must see the identical class layout of the build-only overlay.
foreach(sp_io_consumer io sp_sim_core sp_capture sp_real_aim sp_deploy_protocol_test sp_live_pose_test)
  target_include_directories(${sp_io_consumer} BEFORE PRIVATE "${SP_IO_OVERLAY_ROOT}")
endforeach()
